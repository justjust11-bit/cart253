 * A series of lines across the canvas
 */

"use strict";

/**
 * Creates the canvas
 */
function setup() {
    createCanvas(500, 500);
}

/**
 * Draws lines across the canvas with increasing thickness and
 * gradually lightening colour
 */
function draw() {

  background("pink");
  bkgGradient();
  let x = 0
  let strokeColor = 0
  let lineThick = 0
  let step = 0
  push();
  while (x<= width){
    stroke(strokeColor);
    strokeWeight(lineThick);
    line(x, 0, x, height);
    step +=5
    x+=50;
    x+=step
    strokeColor+=20;
    lineThick+=3;
  }
  pop();

  let y = 0
  let strokeColor2 = 0
  let lineThick2 = 0
  push();
    while (y<= height){
    stroke(strokeColor2);
    strokeWeight(lineThick2);
    line(0, y, width, y);
    y+=50;
    strokeColor2+=20;
    lineThick2+=3;
  }



}
